cmake_minimum_required(VERSION 3.8)
project(wheel_description)
find_package(ament_cmake REQUIRED)
install(DIRECTORY urdf meshes DESTINATION share/${PROJECT_NAME})
# Keep one source copy of the externally supplied model and its relative meshes.
foreach(ROBOT_VERSION "r680_lift_openarm_claw1.3" "r680_lift_openarm_claw1.4")
  set(ROBOT_ASSET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../assets/robots/${ROBOT_VERSION}")
  if(EXISTS "${ROBOT_ASSET_DIR}/r680_lift_openarm_robot_viewer.urdf")
    install(DIRECTORY "${ROBOT_ASSET_DIR}/" DESTINATION share/${PROJECT_NAME}/urdf/${ROBOT_VERSION})
  endif()
endforeach()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
